\section{Document-First Onboarding Flow}

\subsection{Overview}
Revolutionary approach where customers upload documents \textbf{first}, and the system automatically fills the application form based on extracted data.

\subsection{Customer Journey}

\begin{tcolorbox}[colback=codecolor,colframe=tealcolor,title=Customer Journey Flow]

\begin{itemize}[noitemsep]

    \item Data Validation
    \item Cross-Verification

\textbf{Step 3: Auto-Filled Form}
\begin{itemize}[noitemsep]

\textbf{Step 4: Review \& Submit}

\end{itemize}

\section{Automated Decision Engine}

\subsection{Overview}
AI-powered credit decision system that evaluates applications across 8 modules with real-time scoring.

\subsection{Decision Modules}

\begin{longtable}{@{}p{4cm}p{2cm}p{8cm}@{}}
\toprule
\textbf{Module} & \textbf{Weight} & \textbf{Description} \\ \midrule
\endfirsthead
\toprule
\textbf{Module} & \textbf{Weight} & \textbf{Description} \\ \midrule
\endhead
Debt Burden Ratio (DBR) & 55\% & Calculates total monthly obligations vs. income. Considers existing loans, credit cards, and proposed loan. \\
\midrule
Age Assessment & 5\% & Optimal Range: 25-55 years. Scoring based on age brackets. \\
\midrule
City Tier & 5\% & Location-based risk assessment (Tier 1-4 cities). \\
\midrule
Income Verification & 10\% & Salary adequacy check with multiple income brackets. \\
\midrule
SPU Checks & 5\% & 5 Critical Compliance Checks: PEP, SBP Blacklist, NADRA Verisys, Internal Watchlist, CCL. \\
\midrule
EAVMU Investigation & 5\% & Document Authenticity Verification, Customer Contact, Employment Verification, Reference Checks. \\
\midrule
Application Scorecard & 15\% & Form Completeness, Data Consistency, Supporting Documents Quality, Previous Relationship. \\
\midrule
Behavioral Scorecard & 5\% & Transaction History, Account Conduct, Previous Loan Repayment, Cross-Sell Potential (ETB Only). \\
\bottomrule
\caption{Decision Engine Modules}
\end{longtable}

\subsection{Module 1: Debt Burden Ratio (DBR) - Weight: 55\%}

\textbf{Score Calculation:}
\begin{itemize}
    \item DBR $<$ 40\%: 100 points
    \item DBR 40-50\%: 75 points
    \item DBR 50-60\%: 50 points
    \item DBR $>$ 60\%: 0 points (Auto-Reject)
\end{itemize}

\subsection{Decision Output}

\textbf{Final Score Range: 0-100}
\begin{itemize}
    \item \textcolor{green}{\textbf{$\geq$ 70: APPROVE}} (Green)
    \item \textcolor{orange}{\textbf{50-69: REFER}} (Yellow - Manual Review)
    \item \textcolor{red}{\textbf{$<$ 50: REJECT}} (Red)
\end{itemize}

\textbf{Additional Outputs:}
\begin{itemize}[noitemsep]
    \item Risk Level (Low, Medium, High, Critical)
    \item Recommended Loan Amount
    \item Recommended Tenure
    \item Interest Rate Band
    \item Required Conditions (if any)
\end{itemize}

\section{Instant Loan Feature}

\subsection{Overview}
Revolutionary loan product for trusted customers with \textbf{instant approval and disbursement} (no manual intervention).

\subsection{Eligibility Criteria}

\textbf{Must Meet ALL Conditions:}
\begin{enumerate}
    \item Loan Amount: $\leq$ PKR 750,000 (7.5 Lac)
    \item Customer Type: ETB (Existing to Bank) only
    \item Application Source: Mobile App only
    \item All SPU Checks: PASS
    \item Decision Score: $\geq$ 70 (Auto-Approve threshold)
    \item DBR: $<$ 50\%
    \item No Delinquency: 0 days past due on existing accounts
\end{enumerate}

\subsection{Instant Loan Processing Flow}

\begin{tcolorbox}[colback=codecolor,colframe=tealcolor,title=12-Second Processing]
\textbf{Customer Submits Application} (Mobile App) \\
$\downarrow$ (0.5 seconds) \\
\textbf{System Detects:} Amount $\leq$ 7.5 Lac + ETB + Mobile \\
$\downarrow$ (2 seconds) \\
\textbf{Automatic SPU Checks} (PEP, SBP, NADRA, Watchlist, CCL) \\
$\downarrow$ (3 seconds) \\
\textbf{eCIB Fetch \& Analysis} \\
$\downarrow$ (5 seconds) \\
\textbf{Decision Engine Calculation} \\
$\downarrow$ (If Score $\geq$ 70) \\
\textbf{INSTANT APPROVAL} \\
$\downarrow$ (1 second) \\
\textbf{INSTANT DISBURSEMENT} \\
$\downarrow$ \\
\textbf{SMS/Email Notification Sent} \\
$\downarrow$ \\
\textbf{TOTAL TIME: $\sim$12 SECONDS}
\end{tcolorbox}

\subsection{Benefits}

\textbf{For Customers:}
\begin{itemize}
    \item \textbf{12-second approval} (vs. 3-5 days traditional)
    \item \textbf{Immediate fund transfer} to account
    \item \textbf{Fully mobile-based} - no branch visit
    \item \textbf{Minimal documentation} - already on file
    \item \textbf{24/7 availability} - anytime, anywhere
\end{itemize}

\textbf{For Bank:}
\begin{itemize}
